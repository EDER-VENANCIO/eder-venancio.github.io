User Function ExportaDEF()
    Local cArquivo := "\\temp\\exportacao_demo.txt"
    Local cConteudo := "ID;DESCRICAO" + CRLF
    Local nHandle := FCreate(cArquivo)

    If nHandle < 0
        MsgStop("Nao foi possivel criar o arquivo.", "Exportacao")
        Return
    EndIf
    FWrite(nHandle, cConteudo)
    FClose(nHandle)
Return
